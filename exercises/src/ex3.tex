\begin{question}
הוכיחו או הפריכו (מותר להסתמך על כל המשפטים שהוכחו בכיתה ובתרגול):
\begin{parts}
  \item לכל קבוצה לא ריקה של מספרים ממשיים יש לפחות חסם מלעיל אחד או לפחות חסם מלרע אחד.
  \item לכל קבוצה לא ריקה של מספרים ממשיים שיש לה לפחות חסם מלעיל אחד יש חסם תחתון.
  \item לכל קבוצה לא ריקה של מספרים ממשיים שיש לה לפחות חסם מלרע אחד יש לפחות חסם מלעיל אחד.
  \item לכל קבוצה לא ריקה של מספרים רציונליים בעלת מקסימום יש חסם עליון רציונלי.
  \item קיימת קבוצה לא ריקה של מספרים ממשיים אשר החסם העליון שלה שווה לחסם התחתון שלה.
  \item קיימת קבוצה לא ריקה של מספרים ממשיים אשר החסם העליון שלה שווה לחסם התחתון שלה ואשר אין לה מקסימום.
\end{parts}
\end{question}

\begin{question}
תהי $A\subseteq\R$ חסומה מלעיל. נגדיר $B=\set{-a}{a\in A}$.
הוכיחו כי $B$ חסומה מלרע וכן כי $\inf(B)=-\sup(A)$.
\end{question}

\begin{question}
יהיו $A,B$ קבוצות לא ריקות של מספרים ממשיים חסומות מלעיל. נגדיר
\[ A+B=\set{a+b}{a\in A,\ b\in B}. \]
הוכיחו כי $A+B$ חסומה מלעיל וכי $\sup(A+B)=\sup(A)+\sup(B)$.
\end{question}

\begin{question}
לכל אחת מהקבוצות הבאות מצאו את $\max,\min,\sup,\inf$ (הוכיחו).
\begin{parts}
  \item $A=\set{\dfrac14-\left(\dfrac13\right)^n}{n\in\N}$
  \item $B=\set{\dfrac{3n-1}{n}}{n\in\N}$
  \item $C=\set{(-1)^n\cdot n}{n\in\N}$
  \item $D=\set{n+\dfrac1m}{n,m\in\N}$
\end{parts}
\end{question}
