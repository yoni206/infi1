\topic{הבינום של ניוטון}

\begin{question}
פתחו סוגריים ופשטו ככל הניתן:
\begin{parts}
  \item $(x+y^2)^6$
  \item $(x^2+3)^6$
  \item $\left(\dfrac12+\sqrt2\right)^5$
  \item $\left(\sqrt x+\dfrac1x\right)^6$
\end{parts}
\end{question}

\begin{question}
מהו המקדם של $\sqrt x$ בביטוי $\left(\dfrac2x+\sqrt x\right)^7$?
\end{question}

\begin{question}
מהו המקדם של $a^7$ בביטוי $\left(\dfrac34\sqrt[3]{a^2}+\dfrac23\sqrt a\right)^{12}$?
\end{question}
